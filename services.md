## Spectacle .webm to mp4 converter

Service to automatically convert .webms generated by spectacle into .mp4

### Install dependencies

```bash
sudo dnf install inotify-tools ffmpeg -y
```

### Create watcher

```bash
mkdir -p ~/bin
nano ~/bin/webm-to-mp4-watcher.sh
```

```bash
#!/bin/sh

WATCH_DIR="$HOME/Videos/Screencasts"

inotifywait -m -e close_write --format '%w%f' "$WATCH_DIR" |
while read file; do
  [[ "$file" != *.webm ]] && continue

  sleep 1

  ffmpeg -i "$file" \
    -c:v libx264 \
    -preset fast \
    -crf 28 \
    -vf scale=1280:-2 \
    -c:a aac \
    -b:a 96k "${file%.webm}.mp4"
done
```

### Make watcher executable

```bash
chmod +x ~/bin/webm-to-mp4-watcher.sh
```

### Create service

```bash
nano ~/.config/systemd/user/webm-watch.service
```

```bash
[Unit]
Description=WebM to MP4 watcher

[Service]
ExecStart=~/.local/bin/webm-to-mp4-watcher.sh
Restart=always

[Install]
WantedBy=default.target
```

### Autostart said service

```bash
systemctl --user enable --now webm-watch
```

## Brother scanner: convert .tiff to .pdf

### Install dependencies

```bash
sudo dnf install inotify-tools img2pdf -y
```

### Create watcher

```bash
mkdir -p ~/bin
nano ~/bin/brscan-pdf-watch.sh
```

```bash
#!/bin/bash

WATCH_DIR="$HOME/brscan"

mkdir -p "$WATCH_DIR"

inotifywait -m -e close_write "$WATCH_DIR" |
while read path action file; do
    case "$file" in
        *.tif|*.tiff)

            INPUT="$path$file"
            OUTPUT="${INPUT%.*}.pdf"

            img2pdf "$INPUT" -o "$OUTPUT"

            if [ $? -eq 0 ]; then
                rm -f "$INPUT"
                echo "Converted and removed: $INPUT"
            else
                echo "Conversion failed: $INPUT"
            fi
            ;;
    esac
done
```

### Make watcher executable

```bash
chmod +x ~/bin/brscan-pdf-watch.sh
```

### Create service

```bash
nano ~/.config/systemd/user/brscan-pdf-watch.service
```

```bash
[Unit]
Description=Brother Scan TIFF to PDF Converter

[Service]
ExecStart=%h/bin/brscan-pdf-watch.sh
Restart=always

[Install]
WantedBy=default.target
```

### Autostart said service

```bash
systemctl --user enable --now brscan-pdf-watch
```


## Brother scanner: auto start wireless connectivity on startup

### Create service

```bash
sudo nano ~/.config/systemd/user/brscan-autostart.service
```

```bash
[Unit]
Description=Auto start brother brscan-skey service
After=graphical-session.target

[Service]
ExecStart=/usr/bin/brscan-skey
Restart=always

[Install]
WantedBy=default.target
```

### Autostart said service

```bash
systemctl --user daemon-reload
systemctl --user enable brscan-autostart.service
systemctl --user start brscan-autostart.service
```


## Auto force refresh outdated packages in discover

### Create service

```bash
sudo nano /etc/systemd/system/pkcon-refresh.service
```

```bash
[Unit]
Description=Force PackageKit metadata refresh

[Service]
Type=oneshot
ExecStart=/usr/bin/pkcon refresh force
```

### Create timer

```bash
sudo nano /etc/systemd/system/pkcon-refresh.timer
```

```bash
[Unit]
Description=Run PackageKit refresh periodically

[Timer]
OnBootSec=10min
OnUnitActiveSec=6h
Persistent=true

[Install]
WantedBy=timers.target
```

### Enable service

```bash
sudo systemctl daemon-reload
sudo systemctl enable --now pkcon-refresh.timer
```
